} // ── ICAL PARSER ───────────────────────────────────────────────────── function parseIcal( raw: string, platform: string, kid: string, sport: string, familyPhone?: string ): CalEvent[] { const events: CalEvent[] = []; } return events; } // ── OVERLAP DETECTOR ──────────────────────────────────────────────── function eventsOverlap(a: CalEvent, b: CalEvent): boolean { if (a.kid !== b.kid || a.uid === b.uid) return false; } } } // ── FORMAT DATE FOR SMS ───────────────────────────────────────────── function fmtDateTime(d: Date): string { const days = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"]; const months = ["Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"]; } // ── SMART REMINDERS (IN-APP BANNERS) ──────────────────────────────── // Writes today's events as reminder rows so calendar.html can show banners. // Safe to run every 30 min — upsert with ignoreDuplicates prevents double rows. async function generateReminders(supabase: any): Promise { const now = new Date(); const endStr = endDate.toLocaleTimeString("en-US", { hour: "numeric", minute: "2-digit", timeZone: "America/New_York", }); const evName = ev.name || "event"; const lower = evName.toLowerCase(); const isGame = lower.includes("game") || lower.includes("vs") || lower.includes(" @ "); const isTournament = lower.includes("tournament") || lower.includes("tourney"); const verb = isTournament ? "tournament" : isGame ? "game" : "practice"; const loc = ev.location ? ` at ${ev.location}` : ""; const durationMins = (endDate.getTime() - startDate.getTime()) / 60000; let message: string; if (isTournament) { message = `${ev.kid} has a ${verb} today from ${timeStr}–${endStr}${loc}.`; } else { console.log(`generateReminders: wrote ${reminders.length} reminder(s) for ${todayStr}`); } } // ── MAIN FUNCTION ─────────────────────────────────────────────────── Deno.serve(async (_req) => { const supabaseUrl = Deno.env.get("SUPABASE_URL")!;
